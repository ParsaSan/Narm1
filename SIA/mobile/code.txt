import 'dart:convert';
import 'package:http/http.dart' as http;
import 'dashboard_models.dart';

class DashboardApiService {
  // آدرس پایه سرور بک‌اند جاوا
  static const String baseUrl = 'http://localhost:8080/api/dashboard';

  // ۱. دریافت آمار اولیه سامانه
  Future<DashboardStatsModel> fetchStats() async {
    final response = await http.get(Uri.parse('$baseUrl/stats'));
    if (response.statusCode == 200) {
      final data = json.decode(utf8.decode(response.bodyBytes));
      return DashboardStatsModel.fromJson(data);
    } else {
      throw Exception('خطا در دریافت آمار از سرور');
    }
  }

  // ۲. دریافت بنرها
  Future<List<BannerModel>> fetchBanners() async {
    final response = await http.get(Uri.parse('$baseUrl/banners'));
    if (response.statusCode == 200) {
      final List data = json.decode(utf8.decode(response.bodyBytes));
      return data.map((item) => BannerModel.fromJson(item)).toList();
    } else {
      throw Exception('خطا در دریافت بنرها از سرور');
    }
  }

  // ۳. دریافت ۳ درس اخیراً بازدیدشده
  Future<List<RecentCourseModel>> fetchRecentCourses() async {
    final response = await http.get(Uri.parse('$baseUrl/recent-courses'));
    if (response.statusCode == 200) {
      final List data = json.decode(utf8.decode(response.bodyBytes));
      return data.map((item) => RecentCourseModel.fromJson(item)).toList();
    } else {
      throw Exception('خطا در دریافت دروس اخیر از سرور');
    }
  }

  // ۴. ارسال کلیک و بازدید درس به سرور
  Future<void> recordVisit(int courseId, String title, String instructor) async {
    try {
      await http.post(
        Uri.parse('$baseUrl/recent-courses/visit'),
        headers: {'Content-Type': 'application/json; charset=UTF-8'},
        body: json.encode({
          'id': courseId,
          'title': title,
          'instructor': instructor,
        }),
      );
    } catch (_) {}
  }

}

class _CourseListScreenState extends State<CourseListScreen> {
  static const _filters = ['همه', 'تخصصی', 'پایه', 'عمومی'];

  String _selectedFilter = 'همه';
  String _query = '';

  // لیست دروس که از سرور دریافت می‌شود
  List<Course> _allCourses = [];
  bool _isLoading = true;
  String? _errorMessage;

  @override
  void initState() {
    super.initState();
    _fetchCoursesFromServer();
  }

  // دریافت اطلاعات زنده از بک‌اند اسپرینگ‌بوت
  Future<void> _fetchCoursesFromServer() async {
    setState(() {
      _isLoading = true;
      _errorMessage = null;
    });

    try {
      final courses = await CourseApiService.fetchCourses();
      setState(() {
        _allCourses = courses;
        _isLoading = false;
      });
    } catch (e) {
      setState(() {
        _errorMessage = 'خطا در برقراری ارتباط با سرور. آیا بک‌اند روشن است؟';
        _isLoading = false;
      });
    }
  }

  // یکسان‌سازی حروف عربی و فارسی برای جستجوی دقیق‌تر
  String _norm(String s) => s
      .replaceAll('ي', 'ی')
      .replaceAll('ك', 'ک')
      .toLowerCase()
      .trim();

  List<Course> get _filtered {
    final q = _norm(_query);
    return _allCourses.where((c) {
      final okType = _selectedFilter == 'همه' || c.type == _selectedFilter;
      final okSearch = q.isEmpty ||
          _norm(c.title).contains(q) ||
          _norm(c.englishTitle).contains(q) ||
          _norm(c.instructor).contains(q) ||
          _norm(c.code).contains(q);
      return okType && okSearch;
    }).toList();
  }

import 'dart:convert';
import 'package:http/http.dart' as http;
import '../models/student_profile_model.dart';

class ProfileApiService {
  // اندروید امولاتور: 10.0.2.2
  // iOS سیمولاتور / وب / دسکتاپ: localhost
  // گوشی واقعی: IP کامپیوتر، مثلا 192.168.1.5
  static const String baseUrl = 'http://localhost:8080/api/profile';

  static const Duration _timeout = Duration(seconds: 10);

  Future<StudentProfile> getProfile() async {
    try {
      final response = await http.get(Uri.parse(baseUrl)).timeout(_timeout);

      if (response.statusCode == 200) {
        final data = jsonDecode(utf8.decode(response.bodyBytes));
        return StudentProfile.fromJson(data as Map<String, dynamic>);
      }
      throw Exception('خطا در دریافت پروفایل (کد ${response.statusCode})');
    } on Exception catch (e) {
      throw Exception(_cleanMessage(e, 'اتصال به سرور برقرار نشد'));
    }
  }

  Future<StudentProfile> updateProfile(StudentProfile profile) async {
    try {
      final response = await http
          .put(
            Uri.parse(baseUrl),
            headers: {'Content-Type': 'application/json; charset=UTF-8'},
            body: jsonEncode(profile.toUpdateJson()),
          )
          .timeout(_timeout);

      if (response.statusCode == 200) {
        final data = jsonDecode(utf8.decode(response.bodyBytes));
        return StudentProfile.fromJson(data as Map<String, dynamic>);
      }
      if (response.statusCode == 400) {
        throw Exception('اطلاعات واردشده معتبر نیست');
      }
      throw Exception('خطا در ذخیره پروفایل (کد ${response.statusCode})');
    } on Exception catch (e) {
      throw Exception(_cleanMessage(e, 'اتصال به سرور برقرار نشد'));
    }
  }

  String _cleanMessage(Object e, String fallback) {
    final text = e.toString().replaceFirst('Exception: ', '');
    // خطاهای شبکه (SocketException, TimeoutException) پیام فنی دارن
    if (text.contains('SocketException') ||
        text.contains('TimeoutException') ||
        text.contains('ClientException')) {
      return fallback;
    }
    return text;
  }
}

package ir.pgu.user.service;

import ir.pgu.user.dto.RegisterRequest;
import ir.pgu.user.dto.LoginRequest;
import ir.pgu.user.model.User;
import ir.pgu.user.repository.UserRepository;
import org.springframework.security.crypto.bcrypt.BCryptPasswordEncoder;
import org.springframework.stereotype.Service;
import ir.pgu.user.dto.AuthResponse;

@Service
public class AuthService {

    private final UserRepository userRepository;
    private final BCryptPasswordEncoder passwordEncoder;

    public AuthService(UserRepository userRepository,
                       BCryptPasswordEncoder passwordEncoder) {
        this.userRepository = userRepository;
        this.passwordEncoder = passwordEncoder;
    }

    public boolean isStudentNumberTaken(String studentNumber) {
        return userRepository.existsByStudentNumber(studentNumber);
    }

    public void register(RegisterRequest request) {

        if (isStudentNumberTaken(request.getStudentNumber())) {
            throw new RuntimeException("Student number already exists");
        }

        if (request.getPhoneNumber() != null &&
                !request.getPhoneNumber().isBlank()) {

            if (userRepository.existsByPhoneNumber(request.getPhoneNumber())) {
                throw new RuntimeException("Phone number already exists");
            }
        }

        User user = new User();

        user.setFirstName(request.getFirstName());
        user.setLastName(request.getLastName());
        user.setStudentNumber(request.getStudentNumber());
        user.setPhoneNumber(request.getPhoneNumber());

        user.setPassword(passwordEncoder.encode(request.getPassword()));

        userRepository.save(user);
    }

    public AuthResponse login(LoginRequest request) {

        User user = userRepository.findByStudentNumber(request.getIdentifier())
                .or(() -> userRepository.findByPhoneNumber(request.getIdentifier()))
                .orElseThrow(() -> new RuntimeException("User not found"));

        if (!passwordEncoder.matches(
                request.getPassword(),
                user.getPassword())) {

            throw new RuntimeException("Invalid password");
        }

        return new AuthResponse(
                user.getId(),
                user.getFirstName(),
                user.getLastName(),
                user.getStudentNumber()
        );
    }
}
